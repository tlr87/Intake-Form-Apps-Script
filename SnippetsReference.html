Diagnostic Web App Tester

[rd3_test_submit]
WordPress backend diagnostic tool that executes live GET and POST requests to the Google Apps Script deployment URL. Handles cURL redirects (302 Moved Temporarily) properly without losing POST data payload context.
functions.php / Code Snippets (PHP) 
/**
 * RD3 Tech Web App Diagnostic Tester
 * Shortcode: [rd3_test_submit]
 */
add_shortcode('rd3_test_submit', function() {

    $web_app_url = trim('https://script.google.com/macros/s/AKfycbwITHknJ3gd17a_XQWGMNOB5tCo1ZdimLe_UAcuxPuBu4B2ab6Wrz3grnCIeu55yTVXGA/exec');

    $run_test = isset($_POST['run_rd3_diag']) && check_admin_referer('rd3_diag_nonce');

    $test_data = array(
        'hp_website'  => '',
        'website_hp'  => '',
        'name'        => 'Diagnostic Test Lead',
        'email'       => 'test@example.com',
        'phone'       => '021000000',
        'address'     => '123 Test Street, Auckland',
        'userType'    => 'Home user',
        'situation'   => 'Something Broken? — Fix a problem',
        'achievement' => 'Testing WordPress backend connection to Google Apps Script.',
        'timeframe'   => 'As soon as possible'
    );

    /**
     * Standard Google Apps Script Compatible Request Handler
     */
    $execute_gas_request = function($url, $method = 'GET', $body = null, $content_type = null) {

        if (!function_exists('curl_init')) {
            return array(
                'code' => 500,
                'body' => 'PHP cURL extension is not enabled on this server.'
            );
        }

        $ch = curl_init();

        curl_setopt($ch, CURLOPT_URL, $url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);
        curl_setopt($ch, CURLOPT_MAXREDIRS, 5);
        curl_setopt($ch, CURLOPT_TIMEOUT, 25);
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
        curl_setopt(
            $ch,
            CURLOPT_USERAGENT,
            'Mozilla/5.0 (Windows NT 10.0; Win64; x64) RD3-Diagnostic-Client/1.0'
        );

        $headers = array('Expect:');

        if ($method === 'POST') {

            curl_setopt($ch, CURLOPT_POST, true);

            if ($body !== null) {
                curl_setopt($ch, CURLOPT_POSTFIELDS, $body);
            }

            if ($content_type) {
                $headers[] = 'Content-Type: ' . $content_type;
            }
        }

        curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);

        $response_body = curl_exec($ch);
        $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $curl_error = curl_error($ch);

        curl_close($ch);

        if ($curl_error) {
            return array(
                'code' => 0,
                'body' => 'cURL Error: ' . $curl_error
            );
        }

        return array(
            'code' => $http_code,
            'body' => $response_body
        );
    };

    ob_start();

    ?>

    <div style="background: #0f172a; color: #f8fafc; padding: 24px; border-radius: 8px; font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, monospace; max-width: 750px; margin: 20px auto; box-shadow: 0 10px 15px -3px rgba(0,0,0,0.3);">

        <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 16px; border-bottom: 1px solid #334155; padding-bottom: 12px;">

            <h3 style="color: #38bdf8; margin: 0; font-size: 18px;">
                RD3 Apps Script Diagnostic Hub
            </h3>

            <form method="post" style="margin: 0;">

                <?php wp_nonce_field('rd3_diag_nonce'); ?>

                <input type="hidden" name="run_rd3_diag" value="1">

                <button type="submit" style="background: #0284c7; color: #ffffff; border: none; padding: 8px 16px; border-radius: 6px; font-weight: 600; cursor: pointer;">
                    Run Diagnostic Tests
                </button>

            </form>

        </div>

        <p style="background: #1e293b; padding: 10px 14px; border-radius: 6px; word-break: break-all; font-size: 13px; margin-top: 0;">

            <strong style="color: #94a3b8;">
                Target Endpoint:
            </strong>

            <br>

            <code style="color: #38bdf8; font-size: 12px;">
                <?php echo esc_html($web_app_url); ?>
            </code>

        </p>

        <?php if (!$run_test) : ?>

            <p style="color: #94a3b8; font-size: 14px; text-align: center; margin: 30px 0;">

                Click <strong>"Run Diagnostic Tests"</strong> above to re-run the tests with correct redirect handling.

            </p>

        <?php else : 

            $response_get = $execute_gas_request(
                $web_app_url,
                'GET'
            );

            $response_json = $execute_gas_request(
                $web_app_url,
                'POST',
                json_encode($test_data),
                'text/plain;charset=utf-8'
            );

            $response_form = $execute_gas_request(
                $web_app_url,
                'POST',
                http_build_query($test_data),
                'application/x-www-form-urlencoded'
            );

            $render_result = function($title, $res) {

                echo '<h4 style="color: #facc15; margin: 20px 0 8px 0; font-size: 15px;">'
                    . esc_html($title)
                    . '</h4>';

                $code = $res['code'];
                $body = $res['body'];

                $badge_color =
                    ($code >= 200 && $code < 300)
                    ? '#22c55e'
                    : (($code >= 300 && $code < 400)
                        ? '#eab308'
                        : '#ef4444');

                $json_decoded = json_decode($body);

                if (json_last_error() === JSON_ERROR_NONE) {

                    $formatted_body = json_encode(
                        $json_decoded,
                        JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES
                    );

                } else {

                    $formatted_body = $body;

                }

                echo '<div style="font-size: 13px; margin-bottom: 6px;">';

                echo '<strong>HTTP Status Code:</strong> '
                    . '<span style="background: '
                    . $badge_color
                    . '; color: #000; padding: 2px 8px; border-radius: 4px; font-weight: bold;">'
                    . esc_html($code)
                    . '</span>';

                echo '</div>';

                echo '<pre style="background: #1e293b; padding: 12px; border-radius: 6px; overflow-x: auto; color: #a7f3d0; font-size: 12px; margin: 0; max-height: 250px;">'
                    . esc_html($formatted_body)
                    . '</pre>';
            };

            $render_result(
                '1. GET Endpoint Reachability Test',
                $response_get
            );

            $render_result(
                '2. POST JSON Payload Test Results',
                $response_json
            );

            $render_result(
                '3. POST Form URL-Encoded Payload Test Results',
                $response_form
            );

        endif; ?>

    </div>

    <?php

    return ob_get_clean();

});
